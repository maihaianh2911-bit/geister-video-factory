function smoke(t, t0, x, y) {
  if (t < t0) return '';
  let s = '';
  for (let i = 0; i < 6; i++) {
    const p = prog(t, t0 + i * .12, t0 + i * .12 + 1.6);
    if (p <= 0 || p >= 1) continue;
    s += `<circle cx="${x + 30 * Math.sin(p * 7 + i)}" cy="${y - 300 * p}" r="${12 + 30 * p}" fill="#c8d0da" opacity="${.45 * (1 - p)}" filter="url(#blur8)"/>`;
  }
  return s;
}
